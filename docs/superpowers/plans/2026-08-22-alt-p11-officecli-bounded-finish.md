    & git -c credential.helper= `
        -c 'credential.helper=!gh auth git-credential' `
        push --dry-run github-ci "HEAD:refs/heads/$($state.DeliveryBranch)"
    if ($LASTEXITCODE -ne 0) { throw 'GITHUB_PUSH_PREFLIGHT_FAILED' }
    git push --dry-run gitlab "HEAD:refs/heads/$($state.DeliveryBranch)"
    if ($LASTEXITCODE -ne 0) { throw 'GITLAB_PUSH_PREFLIGHT_FAILED' }
    $lastConfirmedStage = 'ref-and-push-preflight-passed'

    # Push GitHub first; ci.yml has no push trigger for codex/**.
    & git -c credential.helper= `
        -c 'credential.helper=!gh auth git-credential' `
        push github-ci "HEAD:refs/heads/$($state.DeliveryBranch)"
    if ($LASTEXITCODE -ne 0) { throw 'GITHUB_FEATURE_PUSH_FAILED' }
    if ((Get-RemoteHead -Remote github-ci -Branch $state.DeliveryBranch -GitHub) -cne
        $state.CandidateSHA) {
        throw 'GITHUB_FEATURE_SHA_MISMATCH'
    }
    $lastConfirmedStage = 'github-ref-pushed'

    # Snapshot IDs instead of comparing local and server timestamps.
    $preRuns = @(
        Invoke-GhJSON -FailureCode 'GITHUB_RUN_SNAPSHOT_FAILED' -Arguments @(
            'run', 'list', '--repo', $state.Repository, '--workflow', 'ci.yml',
            '--branch', $state.DeliveryBranch, '--event', 'workflow_dispatch',
            '--limit', '100', '--json', 'databaseId,headSha'
        )
    )
    $preRunIDs = @($preRuns | ForEach-Object { [long]$_.databaseId })
    $encodedRef = [uri]::EscapeDataString($state.DeliveryBranch)
    $pipelineListURI =
        'https://gitlab.example.invalid/api/v4/projects/1c%2Faisuz%2Fai/pipelines' +
        "?ref=$encodedRef&sha=$($state.CandidateSHA)&source=push&per_page=100"
    $prePipelines = @(
        Invoke-RestMethod -Method Get -Headers $gitlabHeaders -Uri $pipelineListURI
    )
    $prePipelineIDs = @($prePipelines | ForEach-Object { [long]$_.id })
    $lastConfirmedStage = 'provider-ids-snapshotted'

    # One GitLab push, then one immediate GitHub exact-SHA dispatch.
    git push gitlab "HEAD:refs/heads/$($state.DeliveryBranch)"
    if ($LASTEXITCODE -ne 0) { throw 'GITLAB_FEATURE_PUSH_FAILED' }
    if ((Get-RemoteHead -Remote gitlab -Branch $state.DeliveryBranch) -cne
            $state.CandidateSHA -or
        (Get-RemoteHead -Remote github-ci -Branch $state.DeliveryBranch -GitHub) -cne
            $state.CandidateSHA) {
        throw 'ALT_FINISH_REMOTE_SHA_MISMATCH'
    }
    gh workflow run ci.yml `
        --repo $state.Repository `
        --ref $state.DeliveryBranch `
        -f "expected_sha=$($state.CandidateSHA)"
    if ($LASTEXITCODE -ne 0) { throw 'GITHUB_ACTIONS_DISPATCH_FAILED' }
    $lastConfirmedStage = 'gitlab-ref-pushed-github-dispatched'

    # Resolve one new run and one new pipeline absent from the snapshots.
    $identityDeadline = [datetime]::UtcNow.AddMinutes(2)
    do {
        $runs = @(
            Invoke-GhJSON -FailureCode 'GITHUB_RUN_LOOKUP_FAILED' -Arguments @(
                'run', 'list', '--repo', $state.Repository, '--workflow', 'ci.yml',
                '--branch', $state.DeliveryBranch, '--event', 'workflow_dispatch',
                '--limit', '100', '--json', 'databaseId,headSha,url'
            )
        )
        $runMatches = @($runs | Where-Object {
            $_.headSha -ceq $state.CandidateSHA -and
            $preRunIDs -notcontains [long]$_.databaseId
        })
        if ($runMatches.Count -gt 1) { throw 'GITHUB_ACTIONS_RUN_AMBIGUOUS' }
        $pipelines = @(
            Invoke-RestMethod -Method Get -Headers $gitlabHeaders -Uri $pipelineListURI
        )
        $pipelineMatches = @($pipelines | Where-Object {
            $_.sha -ceq $state.CandidateSHA -and $_.ref -ceq $state.DeliveryBranch -and
            $_.source -ceq 'push' -and $prePipelineIDs -notcontains [long]$_.id
        })
        if ($pipelineMatches.Count -gt 1) { throw 'GITLAB_PIPELINE_AMBIGUOUS' }
        if ($runMatches.Count -eq 1) {
            $state.GitHubRunID = [long]$runMatches[0].databaseId
            $state.GitHubRunURL = [string]$runMatches[0].url
        }
        if ($pipelineMatches.Count -eq 1) {
            $state.GitLabPipelineID = [long]$pipelineMatches[0].id
            $state.GitLabPipelineURL = [string]$pipelineMatches[0].web_url
        }
        if (-not $state.GitHubRunID -or -not $state.GitLabPipelineID) {
            Start-Sleep -Seconds 5
        }
    } until (
        ($state.GitHubRunID -and $state.GitLabPipelineID) -or
        [datetime]::UtcNow -ge $identityDeadline
    )
    if (-not $state.GitHubRunID) { throw 'GITHUB_ACTIONS_RUN_IDENTITY_INVALID' }
    if (-not $state.GitLabPipelineID) { throw 'GITLAB_PIPELINE_IDENTITY_INVALID' }
    $lastConfirmedStage = 'remote-identities-selected'
    # Interleaved monitoring keeps both providers visible.
    $githubGreen = $false
    $gitlabGreen = $false
    $remoteDeadline = [datetime]::UtcNow.AddHours(3)
    do {
        $githubRun = Invoke-GhJSON -FailureCode 'GITHUB_RUN_STATUS_FAILED' `
            -Arguments @(
                'api',
                "repos/$($state.Repository)/actions/runs/$($state.GitHubRunID)"
            )
        $gitlabPipeline = Invoke-RestMethod -Method Get -Headers $gitlabHeaders `
            -Uri (
                'https://gitlab.example.invalid/api/v4/projects/' +
                "1c%2Faisuz%2Fai/pipelines/$($state.GitLabPipelineID)"
            )
        if ($githubRun.head_sha -cne $state.CandidateSHA -or
            $githubRun.head_branch -cne $state.DeliveryBranch -or
            $githubRun.event -cne 'workflow_dispatch' -or
            [long]$githubRun.run_attempt -ne 1) {
            throw 'GITHUB_RUN_BINDING_INVALID'
        }
        if ($gitlabPipeline.sha -cne $state.CandidateSHA -or
            $gitlabPipeline.ref -cne $state.DeliveryBranch -or
            $gitlabPipeline.source -cne 'push') {
            throw 'GITLAB_PIPELINE_BINDING_INVALID'
        }
        if ($githubRun.status -ceq 'completed' -and
            $githubRun.conclusion -cne 'success') {
            throw "GITHUB_FEATURE_CI_$($githubRun.conclusion.ToUpperInvariant())"
        }
        if ($gitlabPipeline.status -in @('failed', 'canceled', 'skipped', 'manual')) {
            throw "GITLAB_FEATURE_CI_$($gitlabPipeline.status.ToUpperInvariant())"
        }
        $githubGreen =
            $githubRun.status -ceq 'completed' -and $githubRun.conclusion -ceq 'success'
        $gitlabGreen = $gitlabPipeline.status -ceq 'success'
        if (-not ($githubGreen -and $gitlabGreen)) { Start-Sleep -Seconds 15 }
    } until (
        ($githubGreen -and $gitlabGreen) -or
        [datetime]::UtcNow -ge $remoteDeadline
    )
    if (-not ($githubGreen -and $gitlabGreen)) {
        throw 'ALT_FINISH_REMOTE_TIMEOUT'
    }
    $lastConfirmedStage = 'remote-ci-green'

    # Require the exact six-job GitHub graph and successful critical steps.
    $jobsEnvelope = Invoke-GhJSON -FailureCode 'GITHUB_JOBS_QUERY_FAILED' `
        -Arguments @(
            'api',
            "repos/$($state.Repository)/actions/runs/$($state.GitHubRunID)/jobs?per_page=100"
        )
    $jobs = @($jobsEnvelope.jobs)
    if ([int]$jobsEnvelope.total_count -ne 6 -or $jobs.Count -ne 6) {
        if ($jobs.Count -eq 0) { throw 'GITHUB_ACTIONS_BILLING_BLOCKED' }
        throw 'GITHUB_REQUIRED_JOB_SET_INVALID'
    }
    $build = @($jobs | Where-Object { $_.name -ceq 'build-candidate' })
    $alt = @($jobs | Where-Object { $_.name -ceq 'alt-p11-userspace' })
    if ($build.Count -ne 1 -or $alt.Count -ne 1) {
        throw 'GITHUB_REQUIRED_JOB_IDENTITY_INVALID'
    }
    $native = @{}
    foreach ($runner in @(
        'windows-2025', 'ubuntu-24.04', 'macos-15-intel', 'macos-15'
    )) {
        $matches = @($jobs | Where-Object {
            $_.name -like 'native *' -and
            ((@($_.labels) -contains $runner) -or
             $_.name.StartsWith("native ($runner,", [StringComparison]::Ordinal))
        })
        if ($matches.Count -ne 1) { throw "GITHUB_NATIVE_JOB_$($runner)_INVALID" }
        $native[$runner] = $matches[0]
    }
    foreach ($job in $jobs) {
        if ($job.conclusion -cne 'success' -or
            $job.head_sha -cne $state.CandidateSHA -or
            [long]$job.run_id -ne $state.GitHubRunID) {
            throw 'GITHUB_JOB_RESULT_INVALID'
        }
    }

    Assert-StepSuccess -Job $build[0] `
        -StepName 'Bind trusted dispatch to exact commit' `
        -FailureCode 'GITHUB_EXPECTED_SHA_BINDING_FAILED'
    foreach ($runner in $native.Keys) {
        Assert-StepSuccess -Job $native[$runner] `
            -StepName 'Verify qualified OfficeCLI runtime and MCP handshake' `
            -FailureCode "GITHUB_OFFICECLI_$($runner)_FAILED"
    }
    foreach ($runner in @('macos-15-intel', 'macos-15')) {
        Assert-StepSuccess -Job $native[$runner] `
            -StepName 'Verify OfficeCLI Developer ID signature and notarization policy' `
            -FailureCode "GITHUB_SPCTL_$($runner)_FAILED"
    }

    $altRequiredSteps = @(
        'Run exact candidate in pinned ALT p11 userspace',
        'Authenticate private ALT qualification registry',
        'Verify qualified OfficeCLI in pinned ALT p11 userspace',
        'Remove private ALT registry credentials'
    )
    foreach ($stepName in $altRequiredSteps) {
        Assert-StepSuccess -Job $alt[0] -StepName $stepName `
            -FailureCode "GITHUB_ALT_STEP_FAILED_$stepName"
    }
    $altStepNames = @($alt[0].steps | ForEach-Object { [string]$_.name })
    $altPositions = @(
        $altRequiredSteps | ForEach-Object { [array]::IndexOf($altStepNames, $_) }
    )
    if ($altPositions -contains -1 -or
        -not ($altPositions[0] -lt $altPositions[1] -and
              $altPositions[1] -lt $altPositions[2] -and
              $altPositions[2] -lt $altPositions[3])) {
        throw 'GITHUB_ALT_STEP_ORDER_INVALID'
    }
    # Require exact artifacts and one shared candidate digest.
    $artifactEnvelope = Invoke-GhJSON -FailureCode 'GITHUB_ARTIFACT_QUERY_FAILED' `
        -Arguments @(
            'api',
            "repos/$($state.Repository)/actions/runs/$($state.GitHubRunID)/artifacts?per_page=100"
        )
    $artifacts = @($artifactEnvelope.artifacts)
    $requiredArtifacts = @(
        'candidate-binaries',
        'security-auditor-tool',
        'native-evidence-windows-2025',
        'native-evidence-ubuntu-24.04',
        'native-evidence-macos-15-intel',
        'native-evidence-macos-15',
        'alt-userspace-evidence'
    )
    if ([int]$artifactEnvelope.total_count -ne $requiredArtifacts.Count -or
        $artifacts.Count -ne $requiredArtifacts.Count) {
        throw 'GITHUB_ARTIFACT_SET_INVALID'
    }
    foreach ($name in $requiredArtifacts) {
        $matches = @($artifacts | Where-Object {
            $_.name -ceq $name -and -not $_.expired
        })
        if ($matches.Count -ne 1) { throw "GITHUB_ARTIFACT_$($name)_INVALID" }
    }
    $candidateArtifact = @(
        $artifacts | Where-Object { $_.name -ceq 'candidate-binaries' }
    )[0]
    $candidateDigest = [string]$candidateArtifact.digest
    if ($candidateDigest -notmatch '^sha256:[0-9a-f]{64}$') {
        throw 'GITHUB_CANDIDATE_DIGEST_INVALID'
    }

    $evidenceRoot = Join-Path ([IO.Path]::GetTempPath()) `
        "officecli-exact-$($state.GitHubRunID)"
    if (Test-Path -LiteralPath $evidenceRoot) {
        throw 'ALT_FINISH_EVIDENCE_PATH_EXISTS'
    }
    New-Item -ItemType Directory -Path $evidenceRoot -ErrorAction Stop | Out-Null
    foreach ($name in $requiredArtifacts | Where-Object {
        $_ -like 'native-evidence-*' -or $_ -eq 'alt-userspace-evidence'
    }) {
        $destination = Join-Path $evidenceRoot $name
        New-Item -ItemType Directory -Path $destination -ErrorAction Stop | Out-Null
        gh run download $state.GitHubRunID `
            --repo $state.Repository `
            --name $name `
            --dir $destination
        if ($LASTEXITCODE -ne 0) {
            throw "GITHUB_ARTIFACT_DOWNLOAD_$($name)_FAILED"
        }
    }

    $candidateFiles = @(
        Get-ChildItem -LiteralPath $evidenceRoot -Recurse -File |
            Where-Object { $_.Name -like '*candidate.txt' }
    )
    if ($candidateFiles.Count -ne 5) {
        throw 'GITHUB_CANDIDATE_EVIDENCE_COUNT_INVALID'
    }
    foreach ($file in $candidateFiles) {
        if ((Get-Content -LiteralPath $file.FullName -Raw).Trim() -cne
            "candidate_digest=$candidateDigest") {
            throw 'GITHUB_CANDIDATE_DIGEST_MISMATCH'
        }
    }
    $securityAudits = @(
        Get-ChildItem -LiteralPath $evidenceRoot -Recurse -File |
            Where-Object { $_.Name -like 'security-audit*.json' }
    )
    if ($securityAudits.Count -ne 5) {
        throw 'GITHUB_SECURITY_AUDIT_EVIDENCE_INVALID'
    }

    # Re-hash the Linux asset and prove public smoke precedes private qualification.
    $linuxAssets = @(
        Get-ChildItem -LiteralPath $evidenceRoot -Recurse -File `
            -Filter 'officecli-linux-x64'
    )
    if ($linuxAssets.Count -ne 1 -or $linuxAssets[0].Length -ne 35316133 -or
        (Get-FileHash -LiteralPath $linuxAssets[0].FullName -Algorithm SHA256).Hash.ToLowerInvariant() -cne
            '32ef7a21a54a4ca6c9806bf5e9f3d32bfb1291017329c55044cb2aac71822eb8') {
        throw 'GITHUB_OFFICECLI_LINUX_ASSET_EVIDENCE_INVALID'
    }
    $smokeLogs = @(
        Get-ChildItem -LiteralPath $evidenceRoot -Recurse -File -Filter 'smoke.log'
    )
    if ($smokeLogs.Count -ne 1) { throw 'GITHUB_ALT_SMOKE_LOG_INVALID' }
    $smoke = Get-Content -LiteralPath $smokeLogs[0].FullName -Raw
    $publicMarker =
        'ALT_USERSPACE_COMPATIBLE image=registry.altlinux.org/p11/alt@sha256:4c76520bb4935edf624dde76d5e670d54f40938323b185c4c7270881b71fd8ea'
    $privateMarker = "OFFICECLI_ALT_USERSPACE_COMPATIBLE image=$($state.ImageDigest)"
    $publicIndex = $smoke.IndexOf($publicMarker, [StringComparison]::Ordinal)
    $privateIndex = $smoke.IndexOf($privateMarker, [StringComparison]::Ordinal)
    if ($publicIndex -lt 0 -or $privateIndex -le $publicIndex -or
        $smoke.Contains('not found')) {
        throw 'GITHUB_ALT_RUNTIME_EVIDENCE_INVALID'
    }

    # Two macOS lanes must emit one accepted bounded raw-byte result each.
    $runLogLines = @(& gh run view $state.GitHubRunID --repo $state.Repository --log)
    if ($LASTEXITCODE -ne 0) { throw 'GITHUB_RUN_LOG_FAILED' }
    $runLog = $runLogLines -join "`n"
    $spctlMarkers = [regex]::Matches(
        $runLog,
        '(?m)OFFICECLI_SPCTL_(?:ACCEPTED|BARE_MACHO_NOT_APP)\r?$'
    )
    if ($spctlMarkers.Count -ne 2) { throw 'GITHUB_SPCTL_EVIDENCE_INVALID' }
    $lastConfirmedStage = 'github-evidence-verified'

    # GitLab must have one non-retried successful verify job and an artifact.
    $gitlabJobs = @(
        Invoke-RestMethod -Method Get -Headers $gitlabHeaders `
            -Uri (
                'https://gitlab.example.invalid/api/v4/projects/' +
                "1c%2Faisuz%2Fai/pipelines/$($state.GitLabPipelineID)/jobs" +
                '?include_retried=true&per_page=100'
            )
    )
    if ($gitlabJobs.Count -ne 1) { throw 'GITLAB_VERIFY_JOB_SET_INVALID' }
    $verifyJob = $gitlabJobs[0]
    if ($verifyJob.name -cne 'verify' -or
        $verifyJob.status -cne 'success' -or
        $verifyJob.commit.id -cne $state.CandidateSHA -or
        [long]$verifyJob.pipeline.id -ne $state.GitLabPipelineID -or
        [string]::IsNullOrWhiteSpace($verifyJob.artifacts_file.filename)) {
        throw 'GITLAB_VERIFY_JOB_INVALID'
    }
    $state.GitLabVerifyJobID = [long]$verifyJob.id
    $state.GitLabVerifyURL = [string]$verifyJob.web_url
    $gitlabArtifactZip = Join-Path $evidenceRoot `
        "gitlab-verify-$($state.GitLabVerifyJobID)-artifacts.zip"
    Invoke-WebRequest -Method Get -Headers $gitlabHeaders `
        -Uri (
            'https://gitlab.example.invalid/api/v4/projects/' +
            "1c%2Faisuz%2Fai/jobs/$($state.GitLabVerifyJobID)/artifacts"
        ) `
        -OutFile $gitlabArtifactZip
    if (-not (Test-Path -LiteralPath $gitlabArtifactZip) -or
        (Get-Item -LiteralPath $gitlabArtifactZip).Length -eq 0) {
        throw 'GITLAB_VERIFY_ARTIFACT_DOWNLOAD_FAILED'
    }

    $gitlabArtifactRoot = Join-Path $evidenceRoot 'gitlab-verify-artifacts'
    Expand-Archive -LiteralPath $gitlabArtifactZip `
        -DestinationPath $gitlabArtifactRoot -ErrorAction Stop
    $gitlabDist = Join-Path $gitlabArtifactRoot 'dist'
    $gitlabAuditPath = Join-Path $gitlabDist 'SECURITY-AUDIT.json'
    $gitlabManifestPath = Join-Path $gitlabDist 'SHA256SUMS'
    foreach ($requiredPath in @($gitlabAuditPath, $gitlabManifestPath)) {
        if (-not (Test-Path -LiteralPath $requiredPath -PathType Leaf)) {
            throw 'GITLAB_VERIFY_ARTIFACT_CONTENT_MISSING'
        }
    }
    $gitlabAudit = Get-Content -LiteralPath $gitlabAuditPath -Raw -Encoding UTF8 |
        ConvertFrom-Json
    if ([int]$gitlabAudit.schema_version -ne 1 -or -not $gitlabAudit.passed -or
        $gitlabAudit.commit -cne $state.CandidateSHA -or
        $gitlabAudit.history_scope -cne 'exact_commit_ancestry' -or
        @($gitlabAudit.findings).Count -ne 0) {
        throw 'GITLAB_VERIFY_SECURITY_AUDIT_INVALID'
    }
    $gitlabBinaries = @(
        'teamkit-v0.1.5-windows-amd64.exe',
        'teamkit-v0.1.5-linux-amd64',
        'teamkit-v0.1.5-darwin-amd64',
        'teamkit-v0.1.5-darwin-arm64'
    )
    $manifestLines = @(
        Get-Content -LiteralPath $gitlabManifestPath -Encoding UTF8 |
            Where-Object { -not [string]::IsNullOrWhiteSpace($_) }
    )
    if ($manifestLines.Count -ne $gitlabBinaries.Count) {
        throw 'GITLAB_VERIFY_SHA256SUMS_SET_INVALID'
    }
    foreach ($binary in $gitlabBinaries) {
        $pattern = '^([0-9a-f]{64})  ' + [regex]::Escape($binary) + '$'
        $manifestMatches = @($manifestLines | Where-Object { $_ -cmatch $pattern })
        $binaryPath = Join-Path $gitlabDist $binary
        if ($manifestMatches.Count -ne 1 -or
            -not (Test-Path -LiteralPath $binaryPath -PathType Leaf)) {
            throw 'GITLAB_VERIFY_BINARY_SET_INVALID'
        }
        $expectedHash = [regex]::Match($manifestMatches[0], $pattern).Groups[1].Value
        if ((Get-FileHash -LiteralPath $binaryPath -Algorithm SHA256).Hash.ToLowerInvariant() -cne
            $expectedHash) {
            throw 'GITLAB_VERIFY_BINARY_HASH_MISMATCH'
        }
    }
    $lastConfirmedStage = 'gitlab-evidence-verified'

    # Recheck uniqueness, immutable refs and clean Git after both providers finish.
    $finalRuns = @(
        Invoke-GhJSON -FailureCode 'GITHUB_FINAL_RUN_LOOKUP_FAILED' -Arguments @(
            'run', 'list', '--repo', $state.Repository, '--workflow', 'ci.yml',
            '--branch', $state.DeliveryBranch, '--event', 'workflow_dispatch',
            '--limit', '100', '--json', 'databaseId,headSha,url'
        )
    )
    $finalRunMatches = @($finalRuns | Where-Object {
        $_.headSha -ceq $state.CandidateSHA -and
        $preRunIDs -notcontains [long]$_.databaseId
    })
    $finalPipelines = @(
        Invoke-RestMethod -Method Get -Headers $gitlabHeaders -Uri $pipelineListURI
    )
    $finalPipelineMatches = @($finalPipelines | Where-Object {
        $_.sha -ceq $state.CandidateSHA -and $_.ref -ceq $state.DeliveryBranch -and
        $_.source -ceq 'push' -and $prePipelineIDs -notcontains [long]$_.id
    })
    if ($finalRunMatches.Count -ne 1 -or
        [long]$finalRunMatches[0].databaseId -ne $state.GitHubRunID -or
        $finalPipelineMatches.Count -ne 1 -or
        [long]$finalPipelineMatches[0].id -ne $state.GitLabPipelineID) {
        throw 'ALT_FINISH_LATE_REMOTE_IDENTITY_AMBIGUOUS'
    }
    $finalSelectedRun = Invoke-GhJSON `
        -FailureCode 'GITHUB_FINAL_SELECTED_RUN_QUERY_FAILED' `
        -Arguments @(
            'api',
            "repos/$($state.Repository)/actions/runs/$($state.GitHubRunID)"
        )
    if ([long]$finalSelectedRun.run_attempt -ne 1 -or
        $finalSelectedRun.status -cne 'completed' -or
        $finalSelectedRun.conclusion -cne 'success') {
        throw 'GITHUB_FINAL_RUN_ATTEMPT_INVALID'
    }
    $finalGitLabJobs = @(
        Invoke-RestMethod -Method Get -Headers $gitlabHeaders `
            -Uri (
                'https://gitlab.example.invalid/api/v4/projects/' +
                "1c%2Faisuz%2Fai/pipelines/$($state.GitLabPipelineID)/jobs" +
                '?include_retried=true&per_page=100'
            )
    )
    if ($finalGitLabJobs.Count -ne 1 -or
        [long]$finalGitLabJobs[0].id -ne $state.GitLabVerifyJobID -or
        $finalGitLabJobs[0].name -cne 'verify' -or
        $finalGitLabJobs[0].status -cne 'success') {
        throw 'GITLAB_FINAL_VERIFY_RETRY_INVALID'
    }
    $finalGitHubRef = Get-RemoteHead -Remote github-ci `
        -Branch $state.DeliveryBranch -GitHub
    $finalGitLabRef = Get-RemoteHead -Remote gitlab -Branch $state.DeliveryBranch
    if ($finalGitHubRef -cne $state.CandidateSHA -or
        $finalGitLabRef -cne $state.CandidateSHA) {
        throw 'ALT_FINISH_POST_CI_REF_MISMATCH'
    }
    if (git status --porcelain) { throw 'ALT_FINISH_POST_CI_WORKTREE_DIRTY' }
    $lastConfirmedStage = 'post-ci-evidence-verified'

    $result = [ordered]@{
        candidate_sha           = $state.CandidateSHA
        github_run_id           = $state.GitHubRunID
        github_run_url          = $state.GitHubRunURL
        github_candidate_digest = $candidateDigest
        gitlab_pipeline_id      = $state.GitLabPipelineID
        gitlab_pipeline_url     = $state.GitLabPipelineURL
        gitlab_verify_job_id    = $state.GitLabVerifyJobID
        gitlab_verify_job_url   = $state.GitLabVerifyURL
        qualification_image     = $state.ImageDigest
        publication_run_url     = 'https://github.com/dmitry-m1man/kit-all-team/actions/runs/32541524174'
        evidence_directory      = $evidenceRoot
        conclusion              = 'engineering candidate exact-SHA verified'
    }
}
catch {
    $failureCode = [string]$_.Exception.Message
    if ($failureCode -notmatch '^[A-Za-z][A-Za-z0-9_-]*$') {
        $failureCode = 'ALT_FINISH_EXTERNAL_OPERATION_FAILED'
    } else {
        $failureCode = $failureCode.ToUpperInvariant().Replace('-', '_')
    }
    $failureResult = [ordered]@{
        conclusion             = 'bounded finish stopped'
        candidate_sha          = $state.CandidateSHA
        github_run_id          = $state.GitHubRunID
        github_run_url         = $state.GitHubRunURL
        gitlab_pipeline_id     = $state.GitLabPipelineID
        gitlab_pipeline_url    = $state.GitLabPipelineURL
        qualification_image    = $state.ImageDigest
        failure_code           = $failureCode
        last_confirmed_stage    = $lastConfirmedStage
        source_change_performed = $false
    }
}
finally {
    if ($null -ne $gitlabHeaders) {
        $gitlabHeaders.Clear()
        $gitlabHeaders = $null
    }
    $scopedGitHubToken = $null
    if ($previousGHTokenPresent) {
        $env:GH_TOKEN = $previousGHToken
    } else {
        Remove-Item Env:GH_TOKEN -ErrorAction SilentlyContinue
    }
    if ($previousGitHubTokenPresent) {

    }

}

}
